🐍 Python

🐍 Python - Built-in Support

Installation & Setup

# Python comes with sqlite3 built-in (no installation needed) import sqlite3 print(sqlite3.version) # Check version

Basic Connection & CRUD

import sqlite3 # Connect to database (creates if doesn't exist) conn = sqlite3.connect('app.db') cursor = conn.cursor() # Create table cursor.execute(''' CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL UNIQUE, email TEXT NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) ''') # INSERT cursor.execute('INSERT INTO users (username, email) VALUES (?, ?)', ('alice', 'alice@example.com')) conn.commit() # SELECT cursor.execute('SELECT * FROM users WHERE username = ?', ('alice',)) user = cursor.fetchone() print(user) # UPDATE cursor.execute('UPDATE users SET email = ? WHERE username = ?', ('newemail@example.com', 'alice')) conn.commit() # DELETE cursor.execute('DELETE FROM users WHERE username = ?', ('alice',)) conn.commit() # Close connection conn.close()

Best Practices (Python)

import sqlite3 from contextlib import closing # Use context manager for auto-commit/rollback with sqlite3.connect('app.db') as conn: conn.row_factory = sqlite3.Row # Dict-like access cursor = conn.cursor() # Batch insert with executemany users = [ ('bob', 'bob@example.com'), ('charlie', 'charlie@example.com') ] cursor.executemany('INSERT INTO users (username, email) VALUES (?, ?)', users) # Transaction handling try: cursor.execute('UPDATE users SET email = ? WHERE id = ?', ('new@example.com', 1)) conn.commit() except sqlite3.Error as e: conn.rollback() print(f'Error: {e}') # Access rows as dicts cursor.execute('SELECT * FROM users') for row in cursor.fetchall(): print(row['username'], row['email'])

📦 JavaScript / Node.js

📦 Node.js - better-sqlite3 (Recommended)

Installation

# Install better-sqlite3 (fastest, synchronous) npm install better-sqlite3 # Or install sqlite3 (async, callback-based) npm install sqlite3

Basic Connection & CRUD (better-sqlite3)

const Database = require('better-sqlite3'); // Connect to database const db = new Database('app.db'); // Create table db.exec(` CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL UNIQUE, email TEXT NOT NULL ) `); // INSERT const insert = db.prepare('INSERT INTO users (username, email) VALUES (?, ?)'); insert.run('alice', 'alice@example.com'); // SELECT const selectOne = db.prepare('SELECT * FROM users WHERE username = ?'); const user = selectOne.get('alice'); console.log(user); // SELECT ALL const selectAll = db.prepare('SELECT * FROM users'); const users = selectAll.all(); console.log(users); // UPDATE const update = db.prepare('UPDATE users SET email = ? WHERE username = ?'); update.run('newemail@example.com', 'alice'); // DELETE const del = db.prepare('DELETE FROM users WHERE username = ?'); del.run('alice'); // Close database db.close();

Best Practices (Node.js)

const Database = require('better-sqlite3'); class UserRepository { constructor(dbPath) { this.db = new Database(dbPath); this.initTables(); } initTables() { this.db.exec(` CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL UNIQUE, email TEXT NOT NULL ) `); } create(username, email) { const stmt = this.db.prepare( 'INSERT INTO users (username, email) VALUES (?, ?)' ); return stmt.run(username, email).lastInsertRowid; } findById(id) { const stmt = this.db.prepare('SELECT * FROM users WHERE id = ?'); return stmt.get(id); } // Transaction example batchCreate(users) { const insert = this.db.prepare( 'INSERT INTO users (username, email) VALUES (?, ?)' ); const insertMany = this.db.transaction((users) => { for (const user of users) { insert.run(user.username, user.email); } }); insertMany(users); } close() { this.db.close(); } } // Usage const repo = new UserRepository('app.db'); const userId = repo.create('bob', 'bob@example.com'); console.log(repo.findById(userId));

☕ Java - JDBC Driver

Basic Connection & CRUD

import java.sql.*; public class SQLiteExample { public static void main(String[] args) { String url = "jdbc:sqlite:app.db"; try (Connection conn = DriverManager.getConnection(url)) { // Create table String createTable = "CREATE TABLE IF NOT EXISTS users (" + "id INTEGER PRIMARY KEY AUTOINCREMENT, " + "username TEXT NOT NULL, " + "email TEXT NOT NULL)"; Statement stmt = conn.createStatement(); stmt.execute(createTable); // INSERT with PreparedStatement String insert = "INSERT INTO users (username, email) VALUES (?, ?)"; PreparedStatement pstmt = conn.prepareStatement(insert); pstmt.setString(1, "alice"); pstmt.setString(2, "alice@example.com"); pstmt.executeUpdate(); // SELECT String select = "SELECT * FROM users WHERE username = ?"; pstmt = conn.prepareStatement(select); pstmt.setString(1, "alice"); ResultSet rs = pstmt.executeQuery(); while (rs.next()) { System.out.println("ID: " + rs.getInt("id")); System.out.println("Username: " + rs.getString("username")); System.out.println("Email: " + rs.getString("email")); } } catch (SQLException e) { System.out.println(e.getMessage()); } } }

Best Practices (Java)

import java.sql.*; public class UserRepository { private final String dbUrl; public UserRepository(String dbPath) { this.dbUrl = "jdbc:sqlite:" + dbPath; initDatabase(); } private void initDatabase() { try (Connection conn = getConnection()) { String sql = "CREATE TABLE IF NOT EXISTS users (" + "id INTEGER PRIMARY KEY AUTOINCREMENT, " + "username TEXT NOT NULL UNIQUE, " + "email TEXT NOT NULL)"; conn.createStatement().execute(sql); } catch (SQLException e) { throw new RuntimeException("Failed to initialize database", e); } } private Connection getConnection() throws SQLException { return DriverManager.getConnection(dbUrl); } public void create(String username, String email) { String sql = "INSERT INTO users (username, email) VALUES (?, ?)"; try (Connection conn = getConnection(); PreparedStatement pstmt = conn.prepareStatement(sql)) { pstmt.setString(1, username); pstmt.setString(2, email); pstmt.executeUpdate(); } catch (SQLException e) { throw new RuntimeException("Failed to create user", e); } } public User findById(int id) { String sql = "SELECT * FROM users WHERE id = ?"; try (Connection conn = getConnection(); PreparedStatement pstmt = conn.prepareStatement(sql)) { pstmt.setInt(1, id); ResultSet rs = pstmt.executeQuery(); if (rs.next()) { return new User( rs.getInt("id"), rs.getString("username"), rs.getString("email") ); } return null; } catch (SQLException e) { throw new RuntimeException("Failed to find user", e); } } }

Best Practices (Go)

package main import ( "database/sql" "fmt" _ "github.com/mattn/go-sqlite3" ) type UserRepository struct { db *sql.DB } func NewUserRepository(dbPath string) (*UserRepository, error) { db, err := sql.Open("sqlite3", dbPath) if err != nil { return nil, err } repo := &UserRepository{db: db} if err := repo.init(); err != nil { return nil, err } return repo, nil } func (r *UserRepository) init() error { query := `CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL UNIQUE, email TEXT NOT NULL )` _, err := r.db.Exec(query) return err } func (r *UserRepository) Create(username, email string) (int64, error) { result, err := r.db.Exec( "INSERT INTO users (username, email) VALUES (?, ?)", username, email, ) if err != nil { return 0, err } return result.LastInsertId() } func (r *UserRepository) FindByID(id int64) (*User, error) { var user User err := r.db.QueryRow( "SELECT id, username, email FROM users WHERE id = ?", id, ).Scan(&user.ID, &user.Username, &user.Email) if err == sql.ErrNoRows { return nil, nil } if err != nil { return nil, err } return &user, nil } func (r *UserRepository) Close() error { return r.db.Close() }

📊 VBA (Excel)

📊 VBA - ADO Connection

Setup Requirements

Prerequisites:
• Download SQLite ODBC driver from http://www.ch-werner.de/sqliteodbc/
• Install the driver (32-bit or 64-bit matching your Excel version)
• Enable "Microsoft ActiveX Data Objects" reference in VBA Editor

Basic Connection & CRUD

Sub SQLiteExample() Dim conn As Object Dim rs As Object Dim dbPath As String ' Database path dbPath = ThisWorkbook.Path & "\app.db" ' Create connection Set conn = CreateObject("ADODB.Connection") conn.Open "Driver=SQLite3 ODBC Driver;Database=" & dbPath ' Create table conn.Execute "CREATE TABLE IF NOT EXISTS users (id INTEGER PRIMARY KEY, username TEXT, email TEXT)" ' INSERT conn.Execute "INSERT INTO users (username, email) VALUES ('alice', 'alice@example.com')" ' SELECT Set rs = CreateObject("ADODB.Recordset") rs.Open "SELECT * FROM users", conn ' Loop through results Do While Not rs.EOF Debug.Print rs("username") & " - " & rs("email") rs.MoveNext Loop ' Close rs.Close conn.Close Set rs = Nothing Set conn = Nothing End Sub

Export Excel to SQLite (VBA)

Sub ExportToSQLite() Dim conn As Object Dim ws As Worksheet Dim lastRow As Long Dim i As Long Set ws = ThisWorkbook.Sheets("Sheet1") lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row ' Connect Set conn = CreateObject("ADODB.Connection") conn.Open "Driver=SQLite3 ODBC Driver;Database=" & ThisWorkbook.Path & "\data.db" ' Loop through rows (skip header) For i = 2 To lastRow Dim sql As String sql = "INSERT INTO users (username, email) VALUES ('" & _ ws.Cells(i, 1).Value & "', '" & _ ws.Cells(i, 2).Value & "')" conn.Execute sql Next i conn.Close MsgBox "Data exported successfully!" End Sub

🔷 C# (.NET)

🔷 C# - System.Data.SQLite

Installation (NuGet)

# Install via NuGet Package Manager Install-Package System.Data.SQLite

Basic Connection & CRUD

using System; using System.Data.SQLite; class Program { static void Main() { string connectionString = "Data Source=app.db;Version=3;"; using (var conn = new SQLiteConnection(connectionString)) { conn.Open(); // Create table string createTable = @"CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL, email TEXT NOT NULL )"; using (var cmd = new SQLiteCommand(createTable, conn)) { cmd.ExecuteNonQuery(); } // INSERT using (var cmd = new SQLiteCommand(conn)) { cmd.CommandText = "INSERT INTO users (username, email) VALUES (@username, @email)"; cmd.Parameters.AddWithValue("@username", "alice"); cmd.Parameters.AddWithValue("@email", "alice@example.com"); cmd.ExecuteNonQuery(); } // SELECT using (var cmd = new SQLiteCommand("SELECT * FROM users", conn)) using (var reader = cmd.ExecuteReader()) { while (reader.Read()) { Console.WriteLine($"{reader["username"]} - {reader["email"]}"); } } } } }

<?php try { // Connect to database $pdo = new PDO('sqlite:app.db'); $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION); // Create table $pdo->exec("CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL, email TEXT NOT NULL )"); // INSERT $stmt = $pdo->prepare("INSERT INTO users (username, email) VALUES (?, ?)"); $stmt->execute(['alice', 'alice@example.com']); // SELECT $stmt = $pdo->query("SELECT * FROM users"); while ($row = $stmt->fetch(PDO::FETCH_ASSOC)) { echo "{$row['username']} - {$row['email']}\n"; } } catch (PDOException $e) { echo "Error: " . $e->getMessage(); } ?>

✅ Universal Best Practices

✅ Best Practices for All Languages:
• Always use parameterized queries to prevent SQL injection
• Use connection pooling for multi-threaded applications
• Enable WAL mode for better concurrency
• Close connections properly (use try-finally or RAII patterns)
• Use transactions for batch operations
• Handle errors gracefully with proper exception handling
• Set pragmas for performance (foreign_keys, journal_mode, synchronous)
• Use prepared statements for repeated queries
• Avoid SELECT * in production code
• Implement proper logging for debugging

⚠️ Common Pitfalls

Common Mistakes Across Languages:
• String concatenation for SQL (SQL injection vulnerability)
• Not closing database connections (resource leaks)
• Missing error handling (silent failures)
• Running in auto-commit mode for batch operations (slow performance)
• Not using indexes for frequently queried columns
• Ignoring SQLITE_BUSY errors in multi-threaded apps
• Not setting busy_timeout pragma for concurrent access
• Forgetting to enable foreign keys (disabled by default)
